```python
from dataclasses import dataclass

@dataclass(frozen=True)
class Job:
    name: str
    retries: int = 2

def run(job: Job) -> str:
    for attempt in range(job.retries + 1):
        try:
            return execute(job.name)
        except TimeoutError:
            if attempt == job.retries:
                raise
    raise AssertionError("unreachable")
```
